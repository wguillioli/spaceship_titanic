get_percentiles <- function(column){
  print(quantile(column, 
                 prob = c(0, 0.01, 0.05, 0.10, 0.25, 0.50, 
                          0.75, 0.90, 0.95, 0.99, 1),
                 na.rm = TRUE),
        )
}

# split PassengerId in 2

passenger_df$Group <- str_split_fixed(d$PassengerId, "_", 2)[,1]
passenger_df$GroupNum <- as.integer(str_split_fixed(d$PassengerId, "_", 2)[,2])

# calculate group size and append attribute to d
group_size <- passenger_df %>%
  group_by(Group) %>%
  summarise(GroupSize = max(GroupNum))

passenger_df <- passenger_df %>%
  inner_join(group_size, by = "Group")

passenger_df <- passenger_df %>%
  select(PassengerId, Group, GroupSize)

d <- d %>%
  inner_join(passenger_df, by = "PassengerId")

      Group < 6000 ~ "5k",
      Group < 7000 ~ "6k",

      Group < 9000 ~ "8k",
      Group >= 9000 ~ "9k",
      .default = "Other"

  )
